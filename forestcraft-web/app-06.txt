async function buildPublishFiles(mode){
  const files=[];
  if(mode==='skin'){
    const main=await canvasToBlob(skinComposite(false));
    files.push({role:'main',storage:'skin',filename:'minecraft_skin.png',mime:'image/png',blob:main});
    // Tiny duplicate preview is intentional: gallery can use /files/:id without counting a download.
    files.push({role:'preview',storage:'skin',filename:'minecraft_skin_preview.png',mime:'image/png',blob:main});
  }else if(mode==='item'){
    const main=await animatedBlob(state.item.frames,state.item.size);
    files.push({role:'main',storage:'asset',filename:'item_texture.png',mime:'image/png',blob:main});
    const preview=await canvasToBlob(itemComposite(0));
    files.push({role:'preview',storage:'asset',filename:'item_preview.png',mime:'image/png',blob:preview});
    if(state.item.frames.length>1)files.push({role:'mcmeta',storage:'asset',filename:'item_texture.png.mcmeta',mime:'application/json',blob:mcmetaBlobFor(state.item.fps)});
  }else if(mode==='block'){
    for(const face of ['front','back','left','right','top','bottom']){
      const d=state.block.faces[face];
      const main=await animatedBlob(d.frames,state.block.size);
      files.push({role:'texture',storage:'asset',filename:`block_${face}.png`,mime:'image/png',blob:main});
      if(d.frames.length>1)files.push({role:'mcmeta',storage:'asset',filename:`block_${face}.png.mcmeta`,mime:'application/json',blob:mcmetaBlobFor(state.block.fps)});
    }
    const preview=await canvasToBlob(blockComposite('front',0));
    files.push({role:'preview',storage:'asset',filename:'block_preview.png',mime:'image/png',blob:preview});
  }else if(mode==='model'){
    if(!state.model.bb)throw new Error('先に .bbmodel を読み込んでください。');
    const bbBlob=new Blob([JSON.stringify(state.model.bb,null,2)],{type:'application/json'});
    files.push({role:'bbmodel',storage:'model',filename:'model.bbmodel',mime:'application/json',blob:bbBlob});
    const tex=await modelAnimatedBlob();
    files.push({role:'texture',storage:'model',filename:'model_texture.png',mime:'image/png',blob:tex});
    const preview=await canvasToBlob(modelComposite(0));
    files.push({role:'preview',storage:'model',filename:'model_preview.png',mime:'image/png',blob:preview});
    if(state.model.frames.length>1)files.push({role:'mcmeta',storage:'model',filename:'model_texture.png.mcmeta',mime:'application/json',blob:mcmetaBlobFor(state.model.fps)});
  }
  return files;
}
