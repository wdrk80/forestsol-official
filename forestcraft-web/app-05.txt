function normalizedBounds(ctx,bounds=null){
 const W=ctx.canvas.width,H=ctx.canvas.height;if(!bounds)return{x:0,y:0,w:W,h:H};
 const x=clamp(Math.floor(bounds.x),0,W),y=clamp(Math.floor(bounds.y),0,H),w=clamp(Math.floor(bounds.w),0,W-x),h=clamp(Math.floor(bounds.h),0,H-y);return{x,y,w,h};
}
function floodFill(ctx,sx,sy,erase=false,bounds=null){
 const b=normalizedBounds(ctx,bounds),w=ctx.canvas.width,h=ctx.canvas.height;if(sx<b.x||sy<b.y||sx>=b.x+b.w||sy>=b.y+b.h)return;
 const img=ctx.getImageData(0,0,w,h),d=img.data,i0=(sy*w+sx)*4,target=[d[i0],d[i0+1],d[i0+2],d[i0+3]],rep=erase?[0,0,0,0]:[...tonedColor(state.color,sx,sy),255];if(target.every((v,i)=>v===rep[i]))return;
 const q=[[sx,sy]],seen=new Uint8Array(w*h);while(q.length){const [x,y]=q.pop();if(x<b.x||y<b.y||x>=b.x+b.w||y>=b.y+b.h)continue;const idx=y*w+x;if(seen[idx])continue;seen[idx]=1;const i=idx*4;if(d[i]!==target[0]||d[i+1]!==target[1]||d[i+2]!==target[2]||d[i+3]!==target[3])continue;d[i]=rep[0];d[i+1]=rep[1];d[i+2]=rep[2];d[i+3]=rep[3];q.push([x-1,y],[x+1,y],[x,y-1],[x,y+1]);}ctx.putImageData(img,0,0);
}
